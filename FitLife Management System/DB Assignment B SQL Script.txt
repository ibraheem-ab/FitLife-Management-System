--Tables Creation

create table mem (
member_id varchar2(20) primary key,
fname varchar2(50) not null,
sname varchar2(50) not null,
lname varchar2(50) not null,
bDate date,
jDate date not null,
sex varchar2(7) not null,
membership_type varchar2(100),
check (sex in('m','f','male','female' ))
);



create table trainer(
trainer_id varchar2(20) primary key,
tfname varchar2(50) not null,
tsname varchar2(50) not null,
lname varchar2(50) not null,
salary number(4) not null,
specialty varchar2(100),
check(salary between 1000 AND 8000)
);



create table workout_plan(
plan_id varchar2(20) primary key,
plan_name varchar2(50) not null,
plan_description varchar2(200),
plan_duration varchar2(50),
trainer_id varchar2(20) not null,
foreign key (trainer_id) references trainer(trainer_id) on delete cascade
);



create table subsc(
subsc_id varchar2(20) primary key,
payment_status varchar(10) not null,
start_date date not null,
member_id varchar2(20) not null,
check (payment_status in('paid' ,'unpaid' ,'pending')),
foreign key (member_id) references mem(member_id) on delete cascade
);



create table item(
item_id varchar2(20) primary key,
item_name varchar2(50) not null,
price number(6,2) not null,
cat varchar2(30) default null,
check(price between 10 AND 4999.99),
check (cat in ('Protein' ,'Food' ,'Drinks' ,'Snacks'))
);



create table sale(
sale_id varchar2(20) primary key,
sale_datetime date not null,
quantity number (8) not null,
member_id varchar2(20) not null,
item_id varchar2(20) not null,
total_price number(10,2),
check(quantity between 1 AND 100),
foreign key (member_id) references mem(member_id) on delete cascade,
foreign key (item_id) references item(item_id) on delete cascade);



create table trainer_member(
-- represents the relation between tranier and member tables ,which is it a M-N Relation
trainer_id varchar2(20),
member_id varchar2(20),
primary key (trainer_id, member_id),
foreign key (trainer_id) references trainer(trainer_id) on delete cascade,
foreign key (member_id) references mem(member_id) on delete cascade);



create table workout_member(
-- represents the relation between WorkoutPlan and member tables ,which is it a M-N Relation
plan_id varchar2(20),
member_id varchar2(20),
primary key (plan_id,member_id),
foreign key (plan_id) references workout_plan(plan_id) on delete cascade,
foreign key (member_id) references mem(member_id) on delete cascade
);




create table trainer_phone (
--multivalued attribute (trainer phone number), a trainer may have more than one phone number
tphone varchar2(15),
trainer_id varchar2(20),
primary key(tphone,trainer_id),
foreign key (trainer_id) references trainer(trainer_id) on delete cascade);




create table member_phone(
--multivalued attribute (member phone number), a member may have more than one phone number
mphone varchar2(15),
member_id varchar2(20),
primary key(mphone,member_id),
foreign key (member_id) references mem(member_id) on delete cascade);

--Generate some reports


select * from mem;




select m.member_id, m.fname, m.lname, mp.mphone
from mem m
join member_phone mp on m.member_id= mp.member_id
order by m.member_id;





select t.trainer_id ,t.tfname, t.lname, tp.tphone
from trainer t
join trainer_phone tp on t.trainer_id=tp.trainer_id
order by t.trainer_id;





select wp.plan_id, wp.plan_name, wp.plan_description, wp.plan_duration,
t.tfname, t.lname
from workout_plan wp
join trainer t on wp.trainer_id = t.trainer_id;





select s.sale_id, s.sale_datetime, s.quantity, s.total_price,
m.fname, m.lname, i.item_name, i.price
from sale s
join mem m on s.member_id= m.member_id
join item i on s.item_id= i.item_id
order by s.sale_datetime;





select membership_type, count(*) as  total_members
from mem
group by membership_type;





select i.item_name , sum(s.total_price) as total_sales
from sale s
join item i on s.item_id= i.item_id
group by i.item_name
order by total_sales desc;





select m.fname , m.lname,
t.tfname , t.lname 
from trainer_member tm
join mem m on tm.member_id= m.member_id
join trainer t on tm.trainer_id = t.trainer_id
order by m.fname;





select m.fname, m.lname, wp.plan_name
from workout_member wm
join mem m on wm.member_id = m.member_id
join workout_plan wp on wm.plan_id = wp.plan_id
order by m.fname;
